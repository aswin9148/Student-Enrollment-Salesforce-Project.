DB_NAME = "secms.db"

conn = sqlite3.connect(DB_NAME)
cur = conn.cursor()

# Enable Foreign Keys
cur.execute("PRAGMA foreign_keys = ON")


# ============================================================
# DATABASE TABLES
# ============================================================

cur.executescript("""
CREATE TABLE IF NOT EXISTS admins (
    admin_id INTEGER PRIMARY KEY AUTOINCREMENT,
    username TEXT UNIQUE NOT NULL,
    password TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS students (
    student_id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    email TEXT UNIQUE NOT NULL,
    phone TEXT,
    dob TEXT,
    address TEXT,
    status TEXT DEFAULT 'Inactive'
);

CREATE TABLE IF NOT EXISTS courses (
    course_id INTEGER PRIMARY KEY AUTOINCREMENT,
    title TEXT NOT NULL,
    category TEXT NOT NULL,
    duration TEXT,
    credits INTEGER NOT NULL,
    fee REAL DEFAULT 0,
    description TEXT
);

CREATE TABLE IF NOT EXISTS instructors (
    instructor_id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    instructor_code TEXT UNIQUE NOT NULL,
    phone TEXT,
    email TEXT,
    expertise TEXT
);

CREATE TABLE IF NOT EXISTS enrollments (
    enroll_id INTEGER PRIMARY KEY AUTOINCREMENT,
    student_id INTEGER NOT NULL,
    course_id INTEGER NOT NULL,
    instructor_id INTEGER,
    enrollment_date TEXT NOT NULL,
    status TEXT DEFAULT 'Requested',
    fees_paid INTEGER DEFAULT 0,
    total_amount REAL DEFAULT 0,
    comments TEXT,

    UNIQUE(student_id, course_id),

    FOREIGN KEY(student_id)
        REFERENCES students(student_id)
        ON DELETE CASCADE,

    FOREIGN KEY(course_id)
        REFERENCES courses(course_id)
        ON DELETE CASCADE,

    FOREIGN KEY(instructor_id)
        REFERENCES instructors(instructor_id)
        ON DELETE SET NULL
);
""")

conn.commit()

def login():
    print("\n========== SECMS ADMIN LOGIN ==========")

    username = input("Username: ").strip()
    password = input("Password: ").strip()

    hashed = hash_password(password)

    cur.execute(
        "SELECT * FROM admins WHERE username = ? AND password = ?",
        (username, hashed)
    )

    admin = cur.fetchone()

    if admin:
        print("\nLogin successful!")
        return True

    print("\nInvalid username or password.")
    return False

# AUTO ASSIGN INSTRUCTOR
# ============================================================

def auto_assign_instructor(course_id):

    cur.execute(
        "SELECT category FROM courses WHERE course_id = ?",
        (course_id,)
    )

    result = cur.fetchone()

    if not result:
        return None

    category = result[0].lower()

    if "technical" in category:
        expertise = "Technical"
    elif "language" in category:
        expertise = "Language"
    else:
        expertise = "Non-Technical"

    cur.execute("""
        SELECT instructor_id
        FROM instructors
        WHERE expertise LIKE ?
        LIMIT 1
    """, (f"%{expertise}%",))

    instructor = cur.fetchone()

    if instructor:
        return instructor[0]

    return None

def enroll_student():

    print("\n========== NEW ENROLLMENT ==========")

    sid = input("Student ID: ").strip()
    cid = input("Course ID: ").strip()

    # Check student
    cur.execute(
        "SELECT name FROM students WHERE student_id = ?",
        (sid,)
    )

    student = cur.fetchone()

    if not student:
        print("Student ID not found.")
        return

    # Check course
    cur.execute(
        "SELECT title, fee FROM courses WHERE course_id = ?",
        (cid,)
    )

    course = cur.fetchone()

    if not course:
        print("Course ID not found.")
        return

    # Check duplicate
    cur.execute("""
        SELECT enroll_id
        FROM enrollments
        WHERE student_id = ? AND course_id = ?
    """, (sid, cid))

    if cur.fetchone():
        print("Student already enrolled in this course.")
        return

    instructor_id = auto_assign_instructor(cid)

    comments = input("Comments: ").strip()

    cur.execute("""
        INSERT INTO enrollments
        (
            student_id,
            course_id,
            instructor_id,
            enrollment_date,
            status,
            fees_paid,
            total_amount,
            comments
        )
        VALUES (?, ?, ?, ?, ?, ?, ?, ?)
    """, (
        sid,
        cid,
        instructor_id,
        date.today().isoformat(),
        "Requested",
        0,
        course[1],
        comments
    ))

    conn.commit()

    print("\nEnrollment request created successfully.")

    if instructor_id:
        cur.execute(
            "SELECT name FROM instructors WHERE instructor_id = ?",
            (instructor_id,)
        )

        instructor = cur.fetchone()

        if instructor:
            print(f"Auto Assigned Instructor: {instructor[0]}")
    else:
        print("No suitable instructor found.")


def view_enrollments():

    print("\n========== ENROLLMENTS ==========")

    rows = cur.execute("""
        SELECT
            e.enroll_id,
            s.name,
            c.title,
            i.name,
            e.enrollment_date,
            e.status,
            e.fees_paid,
            e.total_amount,
            e.comments
        FROM enrollments e
        JOIN students s
            ON s.student_id = e.student_id
        JOIN courses c
            ON c.course_id = e.course_id
        LEFT JOIN instructors i
            ON i.instructor_id = e.instructor_id
        ORDER BY e.enroll_id
    """).fetchall()

    if not rows:
        print("No enrollments found.")
        return

    for r in rows:
        fee_status = "Paid" if r[6] else "Pending"

        print("-" * 80)
        print(f"Enrollment ID : {r[0]}")
        print(f"Student       : {r[1]}")
        print(f"Course        : {r[2]}")
        print(f"Instructor    : {r[3] or 'Not Assigned'}")
        print(f"Date          : {r[4]}")
        print(f"Status        : {r[5]}")
        print(f"Fee           : ₹{r[7]:.2f}")
        print(f"Fee Status    : {fee_status}")
        print(f"Comments      : {r[8]}")

def pay_fee():

    print("\n========== FEE PAYMENT ==========")

    eid = input("Enrollment ID: ").strip()

    cur.execute("""
        SELECT enroll_id, total_amount, fees_paid
        FROM enrollments
        WHERE enroll_id = ?
    """, (eid,))

    enrollment = cur.fetchone()

    if not enrollment:
        print("Enrollment not found.")
        return

    if enrollment[2]:
        print("Fees already paid.")
        return

    cur.execute("""
        UPDATE enrollments
        SET fees_paid = 1
        WHERE enroll_id = ?
    """, (eid,))

    conn.commit()

    print(f"₹{enrollment[1]:.2f} payment recorded successfully.")

def approve_enrollment():

    print("\n========== APPROVE ENROLLMENT ==========")

    eid = input("Enrollment ID: ").strip()

    cur.execute("""
        SELECT student_id, status, fees_paid
        FROM enrollments
        WHERE enroll_id = ?
    """, (eid,))

    enrollment = cur.fetchone()

    if not enrollment:
        print("Enrollment not found.")
        return

    if enrollment[1] != "Requested":
        print(f"Current status is {enrollment[1]}.")
        return

    if not enrollment[2]:
        print("Fees must be paid before approval.")
        return

    cur.execute("""
        UPDATE enrollments
        SET status = 'Approved'
        WHERE enroll_id = ?
    """, (eid,))

    cur.execute("""
        UPDATE students
        SET status = 'Active'
        WHERE student_id = ?
    """, (enrollment[0],))

    conn.commit()

    print("Enrollment approved.")
    print("Student status changed to Active.")


def reject_enrollment():

    print("\n========== REJECT ENROLLMENT ==========")

    eid = input("Enrollment ID: ").strip()

    cur.execute("""
        SELECT status
        FROM enrollments
        WHERE enroll_id = ?
    """, (eid,))

    enrollment = cur.fetchone()

    if not enrollment:
        print("Enrollment not found.")
        return

    if enrollment[0] != "Requested":
        print("Only Requested enrollments can be rejected.")
        return

    reason = input("Rejection reason: ").strip()

    cur.execute("""
        UPDATE enrollments
        SET status = 'Rejected',
            comments = ?
        WHERE enroll_id = ?
    """, (reason, eid))

    conn.commit()

    print("Enrollment rejected.")
